def get_risk(rule_id: str) -> dict:
    """Return {why, attack_path} for a rule, or safe defaults."""
    return RISK_CONTEXT.get(rule_id, {
        "why": "This control should be enabled/hardened per the selected security framework.",
        "attack_path": ["Control not satisfied", "Increased attack surface", "Potential compromise"],
    })
